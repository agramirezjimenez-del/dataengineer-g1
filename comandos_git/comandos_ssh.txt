### primero generar la llave ssh
ssh-keygen -t ed25519 -C "agramirezjimenez@gmail.com"

### segundo levantar el servicio ssh en windows
Get-Service -Name ssh-agent | Set-Service -StartupType Manual
Start-Service ssh-agent

### tercero adicionar a mis llaves ssh, la llave generada en el paso 1
ssh-add c:/Users/gabyb/.ssh/id_ed25519
